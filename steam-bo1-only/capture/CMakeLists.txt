cmake_minimum_required(VERSION 3.20)
project(BO1CharacterCapture LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use MSVC Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PORTABLE "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${PORTABLE}/vendor/minhook" minhook EXCLUDE_FROM_ALL)
add_library(BO1CharacterCapture SHARED capture.cpp capture.def)
target_include_directories(BO1CharacterCapture PRIVATE "${PORTABLE}/native/src")
target_link_libraries(BO1CharacterCapture PRIVATE minhook)
add_executable(CaptureBO1 launcher.cpp)
target_include_directories(CaptureBO1 PRIVATE "${PORTABLE}/native/src")
target_link_libraries(CaptureBO1 PRIVATE user32 comdlg32 bcrypt)
foreach(target BO1CharacterCapture CaptureBO1)
 target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${target} PRIVATE /W3 /EHsc)
 target_link_options(${target} PRIVATE /DEBUG:NONE)
endforeach()
