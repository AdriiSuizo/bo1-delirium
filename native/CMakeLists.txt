cmake_minimum_required(VERSION 3.20)
project(BO1DeliriumBO1Motions LANGUAGES C CXX)
enable_language(RC)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../VERSION.txt")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../VERSION.txt" BO1_VERSION LIMIT_COUNT 1)
if(NOT BO1_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)-beta\\.([0-9]+)$")
 message(FATAL_ERROR "VERSION.txt must contain a semantic beta version")
endif()
set(BO1_DISPLAY_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3} Beta ${CMAKE_MATCH_4}")
set(BO1_RC_VERSION "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},${CMAKE_MATCH_4}")
configure_file(src/product_info.hpp.in generated/product_info.hpp @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use MSVC Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../vendor/minhook" minhook EXCLUDE_FROM_ALL)
add_library(DeliriumBO1 SHARED src/adapter.cpp src/adapter.def)
target_link_libraries(DeliriumBO1 PRIVATE minhook)
add_executable(BO1SteamDelirium src/launcher.cpp)
add_executable(BO1DeliriumSetup src/launcher.cpp)
target_compile_definitions(BO1DeliriumSetup PRIVATE BO1_DELIRIUM_INSTALLER)
target_link_libraries(BO1DeliriumSetup PRIVATE user32 comdlg32 bcrypt advapi32 ole32 shell32 uuid)
add_executable(BO1DeliriumUninstall WIN32 src/uninstaller.cpp)
target_link_libraries(BO1DeliriumUninstall PRIVATE shell32 user32)
add_library(BO1DeliriumPlugin SHARED src/plutonium_plugin.cpp)
target_link_libraries(BO1DeliriumPlugin PRIVATE shell32 user32)
add_library(winmm SHARED src/winmm_proxy.cpp src/winmm_proxy.def)
target_link_libraries(winmm PRIVATE shell32)
target_link_libraries(BO1SteamDelirium PRIVATE user32 comdlg32 bcrypt)
add_library(nm108_native_bridge SHARED ../bridge/nm108_native_bridge.cpp ../bridge/bridge.def)
foreach(target DeliriumBO1 BO1SteamDelirium BO1DeliriumSetup BO1DeliriumUninstall BO1DeliriumPlugin nm108_native_bridge winmm)
 set(BO1_RC_TYPE VFT_DLL)
 set(BO1_RC_FILENAME "${target}.dll")
 set(BO1_RC_DESCRIPTION "BO1 Delirium runtime")
 if(target STREQUAL BO1DeliriumSetup)
  set(BO1_RC_TYPE VFT_APP)
  set(BO1_RC_FILENAME "Instalar - Install BO1 Delirium.exe")
  set(BO1_RC_DESCRIPTION "BO1 Delirium - Instalador / Installer")
 elseif(target STREQUAL BO1DeliriumUninstall)
  set(BO1_RC_TYPE VFT_APP)
  set(BO1_RC_FILENAME "Desinstalar - Uninstall BO1 Delirium.exe")
  set(BO1_RC_DESCRIPTION "BO1 Delirium - Desinstalador / Uninstaller")
 elseif(target STREQUAL BO1SteamDelirium)
  set(BO1_RC_TYPE VFT_APP)
  set(BO1_RC_FILENAME "BO1PlutoniumDelirium.exe")
  set(BO1_RC_DESCRIPTION "BO1 Delirium development launcher")
 endif()
 configure_file(src/version_info.rc.in "generated/${target}.rc" @ONLY)
 target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/${target}.rc")
 target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${target} PRIVATE /W3 /EHsc)
 target_link_options(${target} PRIVATE /DEBUG:NONE)
endforeach()

enable_testing()
add_executable(BO1CompatibilityTests tests/compatibility.cpp)
target_include_directories(BO1CompatibilityTests PRIVATE src)
set_property(TARGET BO1CompatibilityTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME BO1Compatibility COMMAND BO1CompatibilityTests)
add_executable(BO1InstallationTests tests/installation.cpp)
target_link_libraries(BO1InstallationTests PRIVATE ole32 shell32 uuid)
target_include_directories(BO1InstallationTests PRIVATE src)
set_property(TARGET BO1InstallationTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME BO1Installation COMMAND BO1InstallationTests)
