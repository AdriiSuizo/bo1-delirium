cmake_minimum_required(VERSION 3.20)
project(BO1DeliriumBO1Motions LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use MSVC Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../vendor/minhook" minhook EXCLUDE_FROM_ALL)
add_library(DeliriumBO1 SHARED src/adapter.cpp src/adapter.def)
target_link_libraries(DeliriumBO1 PRIVATE minhook)
add_executable(BO1SteamDelirium src/launcher.cpp)
add_library(winmm SHARED src/winmm_proxy.cpp src/winmm_proxy.def)
target_link_libraries(BO1SteamDelirium PRIVATE user32 comdlg32 bcrypt)
add_library(nm108_native_bridge SHARED ../bridge/nm108_native_bridge.cpp ../bridge/bridge.def)
foreach(target DeliriumBO1 BO1SteamDelirium nm108_native_bridge winmm)
 target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${target} PRIVATE /W3 /EHsc)
 target_link_options(${target} PRIVATE /DEBUG:NONE)
endforeach()
