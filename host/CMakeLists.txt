cmake_minimum_required(VERSION 3.20)
project(BO1DeliriumAdapter LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The BO1 adapter must be built for Win32; the SDK worker is x64")
endif()
add_library(nm108_native_bridge SHARED nm108_native_bridge.cpp)
target_compile_features(nm108_native_bridge PRIVATE cxx_std_17)
target_compile_definitions(nm108_native_bridge PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
set_property(TARGET nm108_native_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(nm108_native_bridge PRIVATE /O2 /W3)
include(CTest)
add_executable(protocol_tests ../tests/protocol_tests.cpp)
target_include_directories(protocol_tests PRIVATE .)
target_compile_features(protocol_tests PRIVATE cxx_std_17)
target_compile_definitions(protocol_tests PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(protocol_tests PRIVATE /UNDEBUG)
add_test(NAME host_protocol COMMAND protocol_tests)

add_executable(crowd_sdk ../tests/crowd_sdk.cpp)
target_compile_features(crowd_sdk PRIVATE cxx_std_17)
target_compile_definitions(crowd_sdk PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(crowd_sdk PRIVATE /O2)
